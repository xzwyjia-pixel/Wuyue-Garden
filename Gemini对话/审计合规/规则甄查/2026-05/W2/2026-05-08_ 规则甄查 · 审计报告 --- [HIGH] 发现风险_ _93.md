$code = @'
import json

def load_rules(file_path):
    with open(file_path, 'r', encoding='utf-8') as f:
        return json.load(f)

def perform_audit(text, rules):
    results = {"HIGH": [], "MEDIUM": [], "LOW": []}
    risk_matrix = rules.get("risk_levels", {})
    for level, details in risk_matrix.items():
        focus_dict = details.get("focus", {})
        # 确保 focus_dict 是字典再进行迭代
        if isinstance(focus_dict, dict):
            for word, advice in focus_dict.items():
                if word in text:
                    results[level].append({"word": word, "advice": advice})
    return results

def report_audit(results):
    print("\n" + "="*45)
    print("      规则甄查 · 静奢风审计报告 (Pro)      ")
    print("="*45)
    found_any = False
    for level in ["HIGH", "MEDIUM", "LOW"]:
        if results[level]:
            found_any = True
            print(f"\n[{level} 风险检测]")
            for item in results[level]:
                print(f"  > 命中风险: 【{item['word']}】")
                print(f"    优化建议: {item['advice']}")
    if not found_any:
        print("\n✨ 内容精密性极高，符合品牌调性。")
    print("\n" + "="*45 + "\n")

if __name__ == "__main__":
    try:
        rules_data = load_rules('rules.json')
        test_content = "这是全网第一的赚钱秘籍，私信我领链接！"
        audit_res = perform_audit(test_content, rules_data)
        report_audit(audit_res)
    except Exception as e:
        print(f"运行时错误: {e}")
'@
$code | Out-File -FilePath "brand_audit_engine.py" -Encoding utf8

3. 为什么这次会成功？

避开了 Shell 的特殊字符解析：通过 python -c 创建 JSON 绕过了 PowerShell 对 “、” 和 ; 的错误解析。
类型安全检查：我在 perform_audit 函数里增加了 isinstance(focus_dict, dict) 的判断，即便 JSON 格式意外受损，脚本也不会直接崩溃。
