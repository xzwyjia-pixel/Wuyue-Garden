def set_folder_custom_icon(folder_path, icon_path):
    """
    功能 B: 配置 desktop.ini 并设置属性
    """
    ini_path = os.path.join(folder_path, "desktop.ini")
    
    # 写入配置文件内容
    content = [
        "[.ShellClassInfo]",
        f"IconResource={icon_path},0",
        "[ViewState]",
        "Mode=",
        "Vid=",
        "FolderType=Generic"
    ]
    
    with open(ini_path, "w", encoding="gb2312") as f:
        f.write("\n".join(content))
    
    # 设置 desktop.ini 为隐藏和系统文件
    ctypes.windll.kernel32.SetFileAttributesW(ini_path, 0x02 | 0x04)
    # 设置文件夹为只读属性（Windows 显示图标的必要前提）
    ctypes.windll.kernel32.SetFileAttributesW(folder_path, 0x01)
