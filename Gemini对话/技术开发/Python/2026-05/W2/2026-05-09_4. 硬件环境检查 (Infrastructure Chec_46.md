import asyncio
import json
import random
from datetime import datetime
from playwright.async_api import async_playwright

# 哨兵配置：目标公告地址
TARGET_SOURCES = {
    "douyin": "https://www.douyin.com/notice", # 示例地址，执行时动态适配
    "channels": "https://channels.weixin.qq.com/platform/notice"
}

async def platform_sentinel():
    async with async_playwright() as p:
        # 模拟 ThinkPad P15V 常用浏览器环境
        browser = await p.chromium.launch(headless=True)
        context = await browser.new_context(
            user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
        )
        page = await context.new_page()
        
        captured_data = []

        for platform, url in TARGET_SOURCES.items():
            try:
                # 随机延迟，模拟真实人类审计行为 (Anti-Fingerprinting)
                await asyncio.sleep(random.uniform(2, 5))
                await page.goto(url, wait_until="domcontentloaded")
                
                # 提取公告标题、时间及原始文本
                notices = await page.locator(".notice-item, .article-list-item").all()
                
                for notice in notices[:5]: # 仅获取最新的 5 条
                    title = await notice.locator("h3, .title").inner_text()
                    date = await notice.locator(".date, .time").inner_text()
                    link = await notice.get_attribute("href")
                    
                    captured_data.append({
                        "platform": platform,
                        "title": title.strip(),
                        "publish_date": date.strip(),
                        "source_url": link,
                        "capture_time": datetime.now().isoformat()
                    })
            except Exception as e:
                print(f"Error capturing {platform}: {str(e)}")

        # 输出结构化 JSON
        with open("E:/MyCodeProjects/intelligence_raw.json", "w", encoding="utf-8") as f:
            json.dump(captured_data, f, ensure_ascii=False, indent=4)
        
        await browser.close()

if __name__ == "__main__":
    asyncio.run(platform_sentinel())
