## 把浏览器的页面设置为抖音主页。让其自动打开

甄先生，明白。既然要追求**“全自动化侦察”**，我们直接对 v_radar_scanner.py 进行一次热更新，将抖音主页设为默认启动项，并让它在浏览器启动后自动加载。
请在 PowerShell 中直接执行这段代码，它会精密覆盖现有的脚本：
$code = @'
# -*- coding: utf-8 -*-
import asyncio, json
from playwright.async_api import async_playwright
from datetime import datetime

async def run_radar():
    async with async_playwright() as p:
        # 默认打开抖音主页
        target_url = "https://www.douyin.com"
        
        browser = await p.chromium.launch(headless=False)
        context = await browser.new_context()
        page = await context.new_page()
        
        print("\n" + "="*50 + "\n       规则甄查 · 外部情报侦察系统      \n" + "="*50)
        print(f"[系统指令] 正在自动导航至：{target_url}")
        
        # 自动打开主页
        await page.goto(target_url)
        
        print("\n[等待中] 请在浏览器内完成目标导航（或登录）。")
        print("确认数据加载后，回到此处按 Enter 开始审计...")
        
        input()

        print("[执行中] 正在扫描页面反馈流...")
        for _ in range(3):
            await page.mouse.wheel(0, 3000)
            await asyncio.sleep(2)
            
        elements = await page.query_selector_all('p, span, div[class*="content"]')
        data = []
        for el in elements:
            try:
                t = (await el.inner_text()).strip()
                if 10 < len(t) < 500: data.append(t)
            except: continue
        
        captured_data = list(set(data))
        result = {
            'timestamp': datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
            'url': page.url,
            'raw_intelligence': captured_data
        }
        
        with open('intelligence_raw.json', 'w', encoding='utf-8') as jf:
            json.dump(result, jf, ensure_ascii=False, indent=4)
            
        print(f"\n[成功] 任务结束。已捕获 {len(captured_data)} 条情报，存入 intelligence_raw.json")
        await browser.close()

if __name__ == "__main__":
    asyncio.run(run_radar())
'@
