from PIL import Image, ImageDraw, ImageFont
import os

# Define the poem and metadata
title = "嘉蒲清夏"
poem_lines = [
    "嘉岁凝寒雪",
    "蒲风入梦凉",
    "清晖映长昼",
    "夏木自深长"
]
author = "甄先生"

# Image settings
width, height = 600, 1000
background_color = (245, 242, 235)  # Aged paper color
text_color = (40, 40, 40)         # Ink black
seal_color = (180, 40, 40)        # Vermilion seal

# Create image
img = Image.new('RGB', (width, height), color=background_color)
draw = ImageDraw.Draw(img)

# Try to find a Chinese font
font_paths = [
    "/usr/share/fonts/opentype/noto/NotoSerifCJK-Regular.ttc",
    "/usr/share/fonts/truetype/noto/NotoSansCJK-Regular.ttc",
    "/usr/share/fonts/truetype/wqy/wqy-microhei.ttc"
]
font_path = None
for p in font_paths:
    if os.path.exists(p):
        font_path = p
        break

if font_path:
    # Font sizes
    title_font = ImageFont.truetype(font_path, 60)
    body_font = ImageFont.truetype(font_path, 45)
    seal_font = ImageFont.truetype(font_path, 25)

    # Draw Title (Vertical or Centered)
    # Let's do a classic vertical layout
    start_x = width - 150
    curr_y = 100
    for char in title:
        draw.text((start_x, curr_y), char, font=title_font, fill=text_color)
        curr_y += 70

    # Draw Poem Lines (Vertical, right to left)
    curr_x = start_x - 100
    for line in poem_lines:
        curr_y = 150
        for char in line:
            draw.text((curr_x, curr_y), char, font=body_font, fill=text_color)
            curr_y += 60
        curr_x -= 80

    # Draw Author and Seal
    curr_y += 40
    # Red Seal Box
    seal_box_x = curr_x - 10
    seal_box_y = curr_y + 20
    draw.rectangle([seal_box_x, seal_box_y, seal_box_x + 60, seal_box_y + 60], outline=seal_color, width=3)
    # Use only first two chars of author for seal if long, but "甄先生" is 3. Let's use "甄"
    draw.text((seal_box_x + 15, seal_box_y + 10), "甄", font=seal_font, fill=seal_color)

# Save
img.save('jiapu_calligraphy.png')
print("Calligraphy image created: jiapu_calligraphy.png")
