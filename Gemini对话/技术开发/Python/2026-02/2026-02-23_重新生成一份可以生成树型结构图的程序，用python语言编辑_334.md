方案一：纯 Python 终端树形图（推荐首选测试）
这段代码没有任何外部依赖，复制粘贴运行即可。它通过递归算法生成经典的 ASCII/Unicode 树形线条。
class Node:
    """定义树的节点"""
    def __init__(self, name):
        self.name = name
        self.children = []

    def add_child(self, child_node):
        self.children.append(child_node)
        return child_node

def print_tree(node, prefix="", is_last=True, is_root=True):
    """递归打印树形结构"""
    if is_root:
        print(node.name)
        prefix = ""
    else:
        # 根据是否是当前层级的最后一个节点，选择不同的连接符
        connector = "└── " if is_last else "├── "
        print(prefix + connector + node.name)
        # 为子节点准备前缀：如果是最后一个节点，前面留空；否则画一条向下的竖线
        prefix += "    " if is_last else "│   "
    
    child_count = len(node.children)
    for i, child in enumerate(node.children):
        is_last_child = (i == child_count - 1)
        print_tree(child, prefix, is_last_child, is_root=False)

方案二：生成高清图片（使用 Graphviz）
如果你希望把它导出为真实的图片（比如 .png），可以使用 graphviz 库。
环境准备：

需要在电脑上安装 Graphviz 软件底层（Mac 可用 brew install graphviz，Windows 需要去官网下载安装并配置环境变量）。
在 Python 环境中安装第三方库：pip install graphviz。

Python 代码：
from graphviz import Digraph

# 初始化一张有向图，设置从左到右的树形结构 (LR) 或者从上到下 (TB)
dot = Digraph(comment='Naval Life System', format='png')
dot.attr(rankdir='TB', size='10,10') # TB = Top to Bottom
dot.attr('node', shape='box', style='rounded,filled', fontname='Arial', fillcolor='#f9f9f9', color='#333333')
dot.attr('edge', color='#666666')

# --- 节点定义 ---
# 根节点
dot.node('A', '🌟 幸福\n(Happiness)', fillcolor='#ffecb3', fontsize='16', fontname='Arial Bold')

# 一级支柱
dot.node('B', '💰 财富\n(Wealth)', fillcolor='#c8e6c9')
dot.node('C', '🔋 健康\n(Health)', fillcolor='#bbdefb')
dot.node('D', '🤝 良好的人际关系\n(Relationships)', fillcolor='#e1bee7')

# 财富的二级
dot.node('B1', '专长\n(Specific Knowledge)')
dot.node('B2', '杠杆\n(Leverage)')
dot.node('B3', '股权\n(Equity)')

# 健康的二级
dot.node('C1', '生理精力\n(Physical Energy)')
dot.node('C2', '心理平静\n(Mental Peace)')

# 关系的二级
dot.node('D1', '互惠价值\n(Mutual Value)')
dot.node('D2', '高维连接\n(High-level Connections)')
dot.node('D3', '清晰的边界\n(Boundaries)')

# --- 连线建立 ---
dot.edges(['AB', 'AC', 'AD'])
dot.edges(['BB1', 'BB2', 'BB3'])
dot.edges(['CC1', 'CC2'])
dot.edges(['DD1', 'DD2', 'DD3'])

# 渲染并保存图片 (会在当前目录生成一个 naval_system.png 文件)
dot.render('naval_system', view=True)
print("图片已生成并保存为 naval_system.png！")
