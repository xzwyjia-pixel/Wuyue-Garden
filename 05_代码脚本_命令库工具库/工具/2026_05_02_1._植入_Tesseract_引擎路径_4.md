引用上一步的配置\n\nclass ZhenAuditor:\n    \"\"\"甄先生专属审计工具类\"\"\"\n    \n    def __init__(self):\n        # 初始化时自动确认路径配置\n        pytesseract.pytesseract.tesseract_cmd = config.TESSERACT_EXE_PATH\n\n    def preprocess_image(self, img_path):\n        \"\"\"图像预处理：极大提升识别精准度\"\"\"\n        img = cv2.imread(img_path)\n        # 灰度化处理\n        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)\n        # 二值化处理：背景变白，文字变黑\n        _, thresh = cv2.threshold(gray, 150, 255, cv2.THRESH_BINARY)\n        return thresh\n\n    def ocr_scan(self, img_path):\n        \"\"\"核心审计逻辑：一键扫描并返回风险点\"\"\"\n        # 3. 对比违规词库\n        found_risks = [word for word in config.FORBIDDEN_WORDS if word in text]\n        \n        return text.strip(), found_risks\n\n\n第四步：在 main.py 中像软件一样调用\n现在，您不需要再输入任何路径或前提，直接写核心业务逻辑。\nfrom zhen_tools import ZhenAuditor\n\n# 1. 实例化审计工具\nauditor = ZhenAuditor()\n\n# 2. 只要一行代码，即可完成：洗图 -> 识别 -> 对比规则\ncontent, risks = auditor.ocr_scan('cover.jpg')\n\n# 3. 输出结果\nprint(\"--- 甄先生规则甄查报告 ---\")\nprint(f\"识别内容: {content}\")\nif risks:\n    print(f\"🚩 风险预警：检测到违规词 {risks}\")\nelse:\n    print(\"✅ 合规通过\")\n\n\n这样操作的 3 大好处：\n\n一劳永逸：以后每次新建 .py 文件，只需 from zhen_tools import ZhenAuditor 即可。\n极易维护：如果哪天 Tesseract 升级了或路径变了，您只需改动 config.py 里的那一行，所有程序都会自动生效。\n专业标准：这是典型的软件工程开发模式，非常符合您作为项目管理者的专业背景。\n\n建议： 您可以先在文件夹里创建这三个文件，把代码粘贴进去，然后把一张短视频封面重命名为 cover.jpg 放进去运行一下 main.py 看看效果。\n如果在运行过程中出现任何红字报错，请随时告诉我。\n\n---\n*从 Gemini 导出，2026-05*\n\n---\n*Gemini · 审计合规 / 规则甄查 · 2026-05-02*\n"}
