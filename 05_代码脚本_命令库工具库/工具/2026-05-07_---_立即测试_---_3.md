在 Windows 中，修改文件夹图标的核心逻辑是：

在文件夹内创建一个隐藏的 desktop.ini 文件，写入图标路径。
将文件夹本身属性设为“只读”（这是 Windows 识别自定义图标的特殊开关）。
刷新系统缓存。

请在 VS Code 中，将以下代码追加到你的 analyze_folder.py 文件中，或者让 Cline 帮你整合：

1. 修改 Windows 图标的核心函数
import os
import ctypes
import configparser

def set_folder_icon(folder_path, icon_path):
    """
    为指定文件夹设置自定义图标
    :param folder_path: 目标文件夹路径
    :param icon_path: .ico 图标文件的绝对路径
    """
    if not os.path.exists(folder_path):
        return "文件夹不存在"
    if not os.path.exists(icon_path):
        return "图标文件不存在"

    # 1. 确保图标路径是绝对路径（Windows 要求）
    abs_icon_path = os.path.abspath(icon_path)
    
    # 2. 创建 desktop.ini 内容
    ini_path = os.path.join(folder_path, 'desktop.ini')
    
    # 如果已存在，先移除属性以便修改
    if os.path.exists(ini_path):
        ctypes.windll.kernel32.SetFileAttributesW(ini_path, 128) # 128 = Normal

    config = configparser.ConfigParser()
    # Windows 的 desktop.ini 格式比较特殊，手动写入更稳定
    ini_content = [
        '[.ShellClassInfo]',
        f'IconResource={abs_icon_path},0',
        '[ViewState]',
        'Mode=',
        'Vid=',
        'FolderType=Generic'
    ]
    
    with open(ini_path, 'w', encoding='gbk') as f: # Windows 默认通常使用 GBK
        f.write('\n'.join(ini_content))

    # 3. 设置 desktop.ini 为 隐藏+系统 文件
    # 2 = Hidden, 4 = System
    ctypes.windll.kernel32.SetFileAttributesW(ini_path, 2 | 4)

    # 4. 【关键步】设置文件夹为“只读”属性 (Read-only)
    # 这是 Windows 告诉资源管理器去检查 desktop.ini 的信号
    # 1 = Read-only
    ctypes.windll.kernel32.SetFileAttributesW(folder_path, 1)

    # 5. 刷新系统图标缓存
    ctypes.windll.shell32.SHChangeNotify(0x08000000, 0, None, None) # SHCNE_ASSOCCHANGED
    
    return f"成功！图标已设置为: {abs_icon_path}"
