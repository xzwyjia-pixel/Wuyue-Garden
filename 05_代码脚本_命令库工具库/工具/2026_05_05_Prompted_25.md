2. 实战进阶：你的第一个 Python 脚本
既然你已经安装好了 Python 版本，可以尝试运行下面这个简单的脚本（保存为 test_baidu.py），这能帮你直接在 Python 里调用刚刚录制的能力：
from playwright.sync_api import sync_playwright

def run():
    with sync_playwright() as p:
        # 启动 Chromium 浏览器，headless=False 可以让你看到浏览器在跑
        browser = p.chromium.launch(headless=False)
        page = browser.new_page()
        
        # 访问网页
        page.goto("https://www.baidu.com")
        print(f"页面标题是: {page.title()}")
        
        # 截图保存，验证成果
        page.screenshot(path="baidu_success.png")
        
        browser.close()

if __name__ == "__main__":
    run()
