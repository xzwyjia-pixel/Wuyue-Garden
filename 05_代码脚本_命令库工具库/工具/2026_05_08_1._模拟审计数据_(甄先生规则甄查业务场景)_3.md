## 统计图表

import pandas as pd
import matplotlib.pyplot as plt
from openpyxl import Workbook
from openpyxl.chart import PieChart, Reference
from openpyxl.styles import Font, Alignment, PatternFill, Border, Side
import os

# 1. 模拟审计数据 (甄先生规则甄查业务场景)
data = {
    '原始文案': [
        '这是全网第一的赚钱秘籍，绝对不封号！',
        '追求极致的资产增值方案，核心逻辑如下。',
        '点击领取私信我，最快入账方式。',
        '最新上线，绝对超值，私信我获取。',
        '静奢风选品，业内深耕多年，确定性极高。',
        '第一款爆品，封号风险极高，大家快看。',
        '资产增值，商业赋能，极致选品。',
        '赚钱就在今天，私信我，领取链接！'
    ],
    '审计结论': [
        '[HIGH]命中第一, [HIGH]命中封号, [MEDIUM]命中赚钱',
        '合格',
        '[MEDIUM]命中领取, [MEDIUM]命中私信, [HIGH]命中最',
        '[HIGH]命中绝对, [MEDIUM]命中私信',
        '合格',
        '[HIGH]命中第一, [HIGH]命中封号',
        '合格',
        '[MEDIUM]命中赚钱, [MEDIUM]命中私信, [MEDIUM]命中领取, [MEDIUM]命中链接'
    ]
}

df = pd.DataFrame(data)

# 2. 统计风险占比逻辑
def get_max_risk(conclusion):
    if 'HIGH' in conclusion: return '高风险 (HIGH)'
    if 'MEDIUM' in conclusion: return '中风险 (MEDIUM)'
    if 'LOW' in conclusion: return '低风险 (LOW)'
    return '合规 (PASS)'

df['风险等级'] = df['审计结论'].apply(get_max_risk)
risk_counts = df['风险等级'].value_counts().reset_index()
risk_counts.columns = ['风险等级', '频次']

# 3. 创建专业 Excel 报告
file_name = 'Zhen_Audit_Analytics_Report.xlsx'
wb = Workbook()

# --- 工作表 1: 审计明细 ---
ws1 = wb.active
ws1.title = "审计明细数据"
headers = ['原始文案', '审计结论', '风险等级']
ws1.append(headers)

# 样式
header_fill = PatternFill(start_color="0D0D0D", end_color="0D0D0D", fill_type="solid") # 甄先生暗黑静奢风
header_font = Font(color="FFFFFF", bold=True)
center_align = Alignment(horizontal="center", vertical="center")

for cell in ws1[1]:
    cell.fill = header_fill
    cell.font = header_font
    cell.alignment = center_align

for r_idx, row in enumerate(df[['原始文案', '审计结论', '风险等级']].values, 2):
    ws1.append(list(row))
    risk_cell = ws1.cell(row=r_idx, column=3)
    # 颜色区分
    if '高风险' in str(risk_cell.value):
        risk_cell.font = Font(color="FF0000", bold=True)
    elif '中风险' in str(risk_cell.value):
        risk_cell.font = Font(color="E26B0A", bold=True)
    elif '合规' in str(risk_cell.value):
        risk_cell.font = Font(color="00B050", bold=True)

ws1.column_dimensions['A'].width = 50
ws1.column_dimensions['B'].width = 40
ws1.column_dimensions['C'].width = 20
ws1.freeze_panes = "A2"

# --- 工作表 2: 统计看板 ---
ws2 = wb.create_sheet("可视化看板")
ws2.append(['风险维度', '出现频次'])
for row in risk_counts.values:
    ws2.append(list(row))

# 插入饼图
chart = PieChart()
labels = Reference(ws2, min_col=1, min_row=2, max_row=len(risk_counts)+1)
chart_data = Reference(ws2, min_col=2, min_row=1, max_row=len(risk_counts)+1)
chart.add_data(chart_data, titles_from_data=True)
chart.set_categories(labels)
chart.title = "文案合规分布 (规则甄查-甄先生)"
chart.width = 15
chart.height = 10

ws2.add_chart(chart, "D2")

wb.save(file_name)
print(file_name)
