2. 第一阶段：创建核心审计类 (E:\MyCodeProjects\core.py)
请在终端中直接运行以下指令（或者要求 Claude 帮你写）：
import re
import json

class RulesInspector:
    """
    “规则甄查-甄先生”核心引擎
    集成基础关键词、正则模式及语义风险预判
    """
    def __init__(self, rules_file='rules.json'):
        with open(rules_file, 'r', encoding='utf-8') as f:
            self.rules = json.load(f)

    def audit(self, text: str):
        results = []
        for rule in self.rules['blacklist']:
            if re.search(rule['pattern'], text):
                results.append({
                    "level": rule['level'],
                    "reason": rule['reason'],
                    "suggestion": rule['suggestion']
                })
        return results
