💡 最后一个小贴士：如何让它更“干净”？
我注意到你的截图里有 __pycache__ 这种项目。如果你想把这些非代码的“杂物”从 Obsidian 里删掉，只需在 PowerShell 里运行最后一次清理指令：
# 运行此指令删除 Obsidian 里不需要的缓存文件夹索引
$apiKey = "8212fe30470f760008c2bc5a9faa24f07d838267fc995c09b8292a8464ab770c"
$apiVaultUrl = "http://127.0.0.1:27123/vault/Michael_Product"
$headers = @{ "Authorization" = "Bearer $apiKey" }

# 删除无用的缓存笔记
Invoke-RestMethod -Method Delete -Uri "$apiVaultUrl/__pycache__.md" -Headers $headers
Invoke-RestMethod -Method Delete -Uri "$apiVaultUrl/.vs.md" -Headers $headers
Invoke-RestMethod -Method Delete -Uri "$apiVaultUrl/.obsidian.md" -Headers $headers

Write-Host "Done! 你的 Obsidian 关系网现在非常纯净了。" -ForegroundColor Green
